void encontrarDivisores(int num, vector<int>& divisores) {
    for (int i = 1; i <= num; ++i) {
        if (num % i == 0) {
            divisores.push_back(i);
        }
    }
}

int encontrarMCD(int a, int b) {
    if (b == 0)
        return a;
    return encontrarMCD(b, a % b);
}
